function Tstrength = transition(states, statefreq, Hint_param, uw_NV, uw_freq, linewidth)
I = [1 0 0;...
     0 1 0;...
     0 0 1;];
tmp10 = uw_NV(1)\*Hint_param.Sx + uw_NV(2)\*Hint_param.Sy + uw_NV(3)\*Hint_param.Sz;
tmp11 = uw_NV(1)\*Hint_param.Ix + uw_NV(2)\*Hint_param.Iy + uw_NV(3)\*Hint_param.Iz;
Hint = Hint_param.gammaNV \* kron(tmp10, I) + …
&#x9; Hint_param.gammaN \* kron(I, tmp11);
N = length(statefreq);
T = zeros(1, length(uw_freq));
for ii = 1:N
&#x9; initstate = states(:,ii);
&#x9; initfreq = statefreq(ii);
for jj = ii:N
finalstate = states(:,jj);
finalfreq = statefreq(jj);
transition_vec = (conj(finalstate’) \* Hint) \* initstate;
transition_amp = (abs(transition_vec’ \* transition_vec))^2;
transition_shape = Lorentzian(uw_freq, initfreq, finalfreq, linewidth);
T = T + transition_amp\*transition_shape;
end
end
Tstrength = T;
end

function y = Lorentzian(uwfreq, init_freq, final_freq, linewidth)
y = 1./(1 + ((uwfreq - abs(final_freq - init_freq)).^2)/((linewidth/2)^2));
end &#x9;
